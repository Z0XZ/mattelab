Oppgave 1 skal løses uten hjelpemidler. I oppgave 2 og 3 kan du bruke lommeregner.

\oppgave{1}{Ballen}
\del{a)}{blekk}{En ball faller $s(t) = 5t^2$ meter på $t$ sekunder. Finn gjennomsnittsfarten fra $t = 1$ til $t = 3$.}{2.5cm}
\del{b)}{blekk}{Regn ut $\frac{s(3 + h) - s(3)}{h}$ for $h = 0{,}1$ og $h = 0{,}01$. Hva tror du farten er når $t = 3$?}{2.5cm}
\del{c)}{blekk}{Vis at $\frac{s(3 + h) - s(3)}{h} = 30 + 5h$. Hva er $s'(3)$?}{2.5cm}

\oppgave{2}{Teen blir kald}
\del{a)}{blekk}{Temperaturen i en kopp te er $T(t) = 20 + 60 \cdot 0{,}9^t$ grader etter $t$ minutter. Finn den gjennomsnittlige vekstfarten de første fem minuttene.}{2.5cm}
\del{b)}{blekk}{Anslå den momentane vekstfarten når $t = 5$. Bruk $h = 0{,}01$.}{2.5cm}
\del{c)}{blekk}{Hva betyr fortegnet og enheten i svaret ditt? Hvorfor er svaret i b mindre i tallverdi enn svaret i a?}{2.5cm}

\oppgave{3}{Finn et mønster}
\del{a)}{blekk}{La $f(x) = x^3$. Anslå $f'(1)$ og $f'(2)$ med $h = 0{,}001$.}{2.5cm}
\del{b)}{blekk}{Gjett et uttrykk for $f'(x)$. Kontroller gjetningen din i Stigningstallfunksjonen med $x^3 - 3x$: Hva skjer med uttrykket når vi trekker fra $3x$?}{2.5cm}
